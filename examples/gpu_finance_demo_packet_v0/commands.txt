# ashiba-receipt-compiler - GPU Finance Demo Packet
# Run from repo root

set -euo pipefail

# 0. Run baseline tests
PYTHONDONTWRITEBYTECODE=1 python3 test_gpu_acceptance.py
PYTHONDONTWRITEBYTECODE=1 python3 test_gpu_collateral.py

# 1. Receipt 1 - Supported GPU capacity (Lambda A10)
./compile examples/gpu_acceptance_lambda_a10_supported \
  --claim-type gpu_capacity_acceptance --card \
  > examples/gpu_finance_demo_packet_v0/receipt_cards/capacity_supported.txt

./compile examples/gpu_acceptance_lambda_a10_supported \
  --claim-type gpu_capacity_acceptance \
  > examples/gpu_finance_demo_packet_v0/receipt_json/capacity_supported.json

python3 -c 'import json; p="examples/gpu_finance_demo_packet_v0/receipt_json/capacity_supported.json"; d=json.load(open(p)); assert d["verdict"]["status"] == "supported", d["verdict"]'

# 2. Receipt 2 - Unknown MIG ambiguity (GI/CI evidence required)
./compile examples/gpu_acceptance_mig_unknown \
  --claim-type gpu_capacity_acceptance --card \
  > examples/gpu_finance_demo_packet_v0/receipt_cards/mig_unknown.txt

./compile examples/gpu_acceptance_mig_unknown \
  --claim-type gpu_capacity_acceptance \
  > examples/gpu_finance_demo_packet_v0/receipt_json/mig_unknown.json

python3 -c 'import json; p="examples/gpu_finance_demo_packet_v0/receipt_json/mig_unknown.json"; d=json.load(open(p)); assert d["verdict"]["status"] == "unknown", d["verdict"]; assert "GI/CI evidence" in d["verdict"]["basis"], d["verdict"]["basis"]'

# 3. Receipt 3 - Contradicted serial collateral binding (SYNTHETIC)
printf 'NOTE: This receipt was generated from a synthetic adversarial fixture.\nIt does not represent any real asset, operator, or customer.\nIt demonstrates ARC behavior when declared identity conflicts with observed identity.\n---\n[ARC-generated output below]\n\n' \
  > examples/gpu_finance_demo_packet_v0/receipt_cards/serial_contradicted_synthetic.txt

./compile examples/gpu_serial_match_contradicted \
  --claim-type gpu_serial_collateral_match --card \
  >> examples/gpu_finance_demo_packet_v0/receipt_cards/serial_contradicted_synthetic.txt

./compile examples/gpu_serial_match_contradicted \
  --claim-type gpu_serial_collateral_match \
  > examples/gpu_finance_demo_packet_v0/receipt_json/serial_contradicted_synthetic.json

python3 -c 'import json; p="examples/gpu_finance_demo_packet_v0/receipt_json/serial_contradicted_synthetic.json"; d=json.load(open(p)); assert d["verdict"]["status"] == "contradicted", d["verdict"]; assert "Serial set mismatch" in d["verdict"]["basis"], d["verdict"]["basis"]'
